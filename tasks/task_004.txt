   const OAuth2 = google.auth.OAuth2;
   
   class GmailService {
     constructor(clientId, clientSecret, redirectUri) {
       this.oAuth2Client = new OAuth2(clientId, clientSecret, redirectUri);
       this.gmail = null;
     }
     
     generateAuthUrl(scopes) {
       return this.oAuth2Client.generateAuthUrl({
         access_type: 'offline',
         prompt: 'consent',
         scope: scopes
       });
     }
     
     async getTokens(code) {
       const {tokens} = await this.oAuth2Client.getToken(code);
       return tokens;
     }
     
     setCredentials(tokens) {
       this.oAuth2Client.setCredentials(tokens);
       this.gmail = google.gmail({version: 'v1', auth: this.oAuth2Client});
     }
   }

   async getHistoryList(startHistoryId, historyTypes = ['messageAdded', 'messageDeleted', 'labelAdded', 'labelRemoved']) {
     try {
       const response = await this.gmail.users.history.list({
         userId: 'me',

       return {
         history: response.data.history || [],

   async performIncrementalSync(userId) {
     // Get the last known historyId for this user
     const syncData = await this.getUserSyncData(userId);
     
     if (!syncData.historyId) {
       // No history ID - need to do a full sync
       return this.performFullSync(userId);
     }
     
     const result = await this.getHistoryList(syncData.historyId);
     
     if (result.invalidHistoryId) {
       // History ID is too old, fall back to full sync
       return this.performFullSync(userId);
     }
     
     // Process history changes
     for (const historyItem of result.history) {
       await this.processHistoryItem(historyItem, userId);
     }
     
     // Update the stored historyId
     await this.updateUserSyncData(userId, result.historyId);
     
     return {
       success: true,
       historyId: result.historyId,
       changesProcessed: result.history.length
     };
   }
